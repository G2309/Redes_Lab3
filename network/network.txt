import asyncio
import redis.asyncio as redis

HOST = "lab3.redesuvg.cloud"
PORT = 6379
PWD  = "UVGRedis2025"

class NetworkNodePubSub:
    def __init__(self, node_id, channels):
        self.node_id = node_id
        self.channels = channels  # lista de canales a suscribirse
        self.algorithm_logic = None

        self.redis = redis.Redis(host=HOST, port=PORT, password=PWD)
        self.pubsub = self.redis.pubsub()
    
    def set_algorithm(self, algorithm):
        self.algorithm_logic = algorithm

    async def listen(self):
        """Escuchar mensajes de Redis y pasarlos al algoritmo"""
        await self.pubsub.subscribe(*self.channels)
        print(f"[{self.node_id}] Suscrito a canales: {self.channels}")

        while True:
            message = await self.pubsub.get_message(ignore_subscribe_messages=True, timeout=1.0)
            if message is not None:
                data = message["data"]
                if isinstance(data, bytes):
                    data = data.decode()
                # Pasar al algoritmo
                if self.algorithm_logic:
                    await self.algorithm_logic.handle_message(data)
            await asyncio.sleep(0.01)  # evitar busy-wait

    async def publish(self, channel, message):
        """Enviar mensaje a un canal"""
        await self.redis.publish(channel, message)
